import re

def normalize_for_tts(text: str) -> str:
    """يصلح نطق الأرقام العشرية والمصطلحات قبل ما تدخل على الـ TTS."""
    # 1. تحويل النسب العشرية: 2.42% -> 2 point 42 percent
    text = re.sub(
        r'(\d+)\.(\d+)\s*%',
        lambda m: f"{m.group(1)} point {' '.join(m.group(2))} percent",
        text
    )
    text = re.sub(r'(\d+)\s*%', r'\1 percent', text)

    # 2. تصليح نطق الاختصارات الفنية
    replacements = {
        r'\bEMAs\b': 'E-M-As',
        r'\bEMA\s*(\d+)\b': r'E-M-A \1',
        r'\b4H\b': '4 hour',
        r'\b1H\b': '1 hour',
        r'\bGC=F\b': 'gold futures',
        r'\bSI=F\b': 'silver',
        r'\bMTF\b': 'multi-timeframe',
        r'\bFed\b': 'Fed',
    }
    for pattern, repl in replacements.items():
        text = re.sub(pattern, repl, text)

    return text
