// Converts a string input into a line style.

bv_fLineStyle(string bv_s) =>

    bv_s == "Solid" ? line.style_solid : bv_s == "Dashed" ? line.style_dashed : line.style_dotted



bv_fFmt(float bv_v) =>

    float bv_av = math.abs(bv_v)

    string bv_suf = bv_av >= 1e9 ? "B" : bv_av >= 1e6 ? "M" : bv_av >= 1e3 ? "K" : ""

    float bv_div = bv_av >= 1e9 ? 1e9 : bv_av >= 1e6 ? 1e6 : bv_av >= 1e3 ? 1e3 : 1.0

    string bv_s = str.tostring(bv_v / bv_div, bv_av >= 1e3 ? "#.##" : "#")

    bv_s + bv_suf



bv_fPctStr(float bv_pct01) =>

    float bv_p = math.max(0.0, math.min(1.0, bv_pct01)) * 100.0

    str.tostring(bv_p, "#.#") + "%"



// Sums bullish sub-bar volume from a lower timeframe.

bv_fBullVolSum(string bv_ltf) =>

    bv_arr = request.security_lower_tf(syminfo.tickerid, bv_ltf, close > open ? volume : 0.0)

    float sum = 0.0

    int bv_n = array.size(bv_arr)

    if bv_n > 0

        for bv_i = 0 to bv_n - 1

            sum += array.get(bv_arr, bv_i)

    sum



// Sums bearish sub-bar volume from a lower timeframe.

bv_fBearVolSum(string bv_ltf) =>

    bv_arr = request.security_lower_tf(syminfo.tickerid, bv_ltf, close < open ? volume : 0.0)

    float sum = 0.0

    int bv_n = array.size(bv_arr)

    if bv_n > 0

        for bv_i = 0 to bv_n - 1

            sum += array.get(bv_arr, bv_i)

    sum



bv_tfResolve(string bv_tfIn) =>

    string bv_tf = bv_tfIn == "" ? timeframe.period : bv_tfIn

    int bv_chartSec = timeframe.in_seconds(timeframe.period)

    int bv_wantSec = timeframe.in_seconds(bv_tf)

    (not na(bv_chartSec) and not na(bv_wantSec) and bv_wantSec < bv_chartSec) ? timeframe.period : bv_tf

        bv_swCur.bv_h := bv_pxNew(high, time)

    if na(bv_swCur.bv_l.bv_p) or low < bv_swCur.bv_l.bv_p

        bv_swCur.bv_l := bv_pxNew(low, time)

    float bv_candL = bv_cand.bv_l.bv_p



    bool bv_isPh = not na(bv_candH)

        if bv_isPh and not na(bv_b.bv_h.bv_p) and bv_candH <= bv_b.bv_h.bv_p

            bv_isPh := false

        if bv_isPl and not na(bv_b.bv_l.bv_p) and bv_candL >= bv_b.bv_l.bv_p

// LTF bull/bear volume.
